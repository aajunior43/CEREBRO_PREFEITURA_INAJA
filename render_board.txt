        function renderBoard() {
            if (!Array.isArray(tasks)) tasks = [];
            const cols = ['todo', 'in-progress', 'done'];
            const visible = _getVisibleTasks();
            cols.forEach(col => {
                const body = document.getElementById(`body-${col}`);
                const badge = document.getElementById(`badge-${col}`);
                if (!body || !badge) {
                    throw new Error(`Estrutura da coluna "${col}" não encontrada.`);
                }
                const colTasks = _sortTasks(visible.filter(t => t.status === col));
                badge.textContent = colTasks.length;
                body.innerHTML = '';
                if (!colTasks.length) {
                    body.innerHTML = '<div class="kb-empty">Nenhuma tarefa aqui.<br>Arraste uma card ou crie uma nova.</div>';
                }
                colTasks.forEach(task => {
                    try {
                        body.appendChild(createCard(task));
                    } catch (e) {
                        console.error('[KANBAN] Erro ao renderizar card:', task, e);
                    }
                });
            });
            updateStats();
            updateCatCounts();
            if (currentView === 'list') renderListView();
        }

        function _dueBadge(dateStr) {
            if (!dateStr) return '';
            const today = new Date(); today.setHours(0,0,0,0);
            const due = new Date(dateStr + 'T00:00:00');
            const diff = Math.round((due - today) / 86400000);
            let cls, label;
            if (diff < 0)       { cls = 'kb-due-overdue'; label = `⚠️ Vencido (${Math.abs(diff)}d)`; }
            else if (diff === 0){ cls = 'kb-due-today';   label = '⏰ Hoje'; }
            else if (diff <= 3) { cls = 'kb-due-today';   label = `📅 ${diff}d`; }
            else                { cls = 'kb-due-ok';      label = `📅 ${due.toLocaleDateString('pt-BR',{day:'2-digit',month:'2-digit'})}`; }
            return `<span class="kb-due ${cls}">${label}</span>`;
        }

        function _catBadge(cat) {
            if (!cat) return '';
            const map = {
                EMPENHAR: 'kb-cat-empenhar', LIQUIDAR: 'kb-cat-liquidar',
                PAGAR: 'kb-cat-pagar', OBS: 'kb-cat-obs',
                RESOLVER: 'kb-cat-resolver', OUTROS: 'kb-cat-outros'
            };
            const cls = map[cat.toUpperCase()] || 'kb-cat-outros';
            return `<span class="kb-cat ${cls}">${cat.toUpperCase()}</span>`;
        }

        function updateCatCounts() {
            const cats = ['all','empenhar','liquidar','pagar','obs','resolver','outros'];
            cats.forEach(c => {
                const el = document.getElementById(`cnt-cat-${c}`);
                if (!el) return;
                const count = c === 'all' ? tasks.length : tasks.filter(t => (t.categoria||'').toUpperCase() === c.toUpperCase()).length;
                el.textContent = count > 0 ? `(${count})` : '';
            });
        }

        function setCatFilter(cat) {
            activeCatFilter = cat;
            document.querySelectorAll('.kb-cat-filter-btn').forEach(btn => btn.classList.remove('active'));
            const btns = document.querySelectorAll('.kb-cat-filter-btn');
            btns.forEach(btn => {
                const txt = btn.textContent.trim().split('(')[0].trim();
                if (cat === '' && btn.classList.contains('kb-cat-filter-all')) btn.classList.add('active');
                else if (cat && txt === cat) btn.classList.add('active');
            });
            renderBoard();
        }

        function onSearch() {
            activeSearch = (document.getElementById('kb-search').value || '').trim();
            renderBoard();
        }

        function _avatarBadge(name) {
            if (!name) return '';
            const initials = name.trim().split(/\s+/).map(w => w[0]).slice(0,2).join('').toUpperCase();
            const colors = ['#0891b2','#d97706','#059669','#dc2626','#7c3aed','#db2777'];
            const color = colors[name.charCodeAt(0) % colors.length];
            return `<span class="kb-avatar" style="background:${color}" title="${esc(name)}">${initials}</span>`;
        }

        function _quickMoveLabel(status) {
            if (status === 'todo') return { label: '→ Em andamento', next: 'in-progress', cls: '' };
            if (status === 'in-progress') return { label: '→ Concluído', next: 'done', cls: 'kb-quick-move-done' };
            return null;
        }

        function createCard(task) {
            const card = document.createElement('div');
            card.className = `kb-card kb-prio-border-${task.priority}`;
            card.draggable = true;
            card.dataset.id = task.id;
            const prioLabel = { high: '🔴 Alta', medium: '🟡 Média', low: '🟢 Baixa' };
            const prioClass = { high: 'kb-prio-high', medium: 'kb-prio-medium', low: 'kb-prio-low' };
            const attachments = Array.isArray(task.attachments) ? task.attachments : [];
            const qm = _quickMoveLabel(task.status);
            const quickMoveBtn = qm ? `<button class="kb-quick-move ${qm.cls}" onclick="quickMove('${task.id}','${qm.next}')" title="Mover para próxima etapa">${qm.label}</button>` : '';
            card.innerHTML = `
      <div class="kb-card-top">
        <div class="kb-card-title">${esc(task.title)}</div>
        <div class="kb-card-actions">
          ${_avatarBadge(task.responsavel)}
          <button class="kb-icon-btn" onclick="openEdit('${task.id}')" title="Editar">✏️</button>
          <button class="kb-icon-btn" onclick="breakdownTaskWithAi('${task.id}')" title="Quebrar em subtarefas com IA">✨</button>
          <button class="kb-icon-btn" style="color:#ef4444" onclick="confirmDelete('${task.id}')" title="Excluir">🗑</button>
        </div>
      </div>
      ${task.description ? `<div class="kb-card-desc">${esc(task.description)}</div>` : ''}
      ${attachments.length ? `<div class="kb-attachments">${attachments.map(att => `
        <span class="kb-attachment-chip">
          <a href="/api/kanban/${task.id}/attachments/${att.id}/download" target="_blank" rel="noopener">📎 ${esc(att.file_name)}</a>
          <button class="kb-attachment-del" onclick="deleteAttachment('${task.id}', ${att.id})" title="Excluir anexo">✕</button>
        </span>`).join('')}</div>` : ''}
      <div class="kb-card-foot">
        <span class="kb-priority ${prioClass[task.priority]}">${prioLabel[task.priority]}</span>
        ${_catBadge(task.categoria)}
        ${_dueBadge(task.data_vencimento)}
        ${quickMoveBtn}
      </div>`;
            card.addEventListener('dragstart', e => { dragTaskId = task.id; card.classList.add('dragging'); e.dataTransfer.effectAllowed = 'move'; });
            card.addEventListener('dragend', () => { dragTaskId = null; card.classList.remove('dragging'); document.querySelectorAll('.kb-col-body').forEach(b => b.classList.remove('drag-over')); });
            card.addEventListener('dragover', e => {
                if (e.dataTransfer?.types?.includes('Files')) {
                    e.preventDefault();
                    card.classList.add('drag-over');
                }
            });
            card.addEventListener('dragleave', () => card.classList.remove('drag-over'));
            card.addEventListener('drop', async e => {
                if (!e.dataTransfer?.files?.length) return;
                e.preventDefault();
                e.stopPropagation();
                card.classList.remove('drag-over');
                const file = e.dataTransfer.files[0];
                try {
                    await uploadAttachment(task.id, file);
                    await loadAll();
                    showToast('Arquivo anexado à tarefa', 'success');
                } catch (err) {
                    showToast(err.message || 'Erro ao anexar arquivo', 'error');
                }
            });
            return card;

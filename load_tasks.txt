        function _getVisibleTasks() {
            let list = activeCatFilter ? tasks.filter(t => (t.categoria||'').toUpperCase() === activeCatFilter) : tasks;
            if (activeSearch) {
                const q = activeSearch.toLowerCase();
                list = list.filter(t => (t.title||'').toLowerCase().includes(q) || (t.description||'').toLowerCase().includes(q));
            }
            return list;
        }

        function _sortTasks(list) {
            const sortBy = (document.getElementById('kb-sort') || {}).value || 'priority';
            const prioOrder = { high: 3, medium: 2, low: 1 };
            return [...list].sort((a, b) => {
                if (sortBy === 'priority') return (prioOrder[b.priority]||0) - (prioOrder[a.priority]||0);
                if (sortBy === 'due') {
                    const ad = a.data_vencimento || '9999-12-31';
                    const bd = b.data_vencimento || '9999-12-31';
                    return ad < bd ? -1 : ad > bd ? 1 : 0;
                }
                if (sortBy === 'created') return (b.criado_em||'') < (a.criado_em||'') ? -1 : 1;
                if (sortBy === 'title') return (a.title||'').localeCompare(b.title||'', 'pt-BR');
                return 0;
            });
        }
